/// The editor background every ratio is measured against.
const BG: &str = "#111214";

/// Relative luminance, as WCAG 2.1 defines it.
fn luminance(hex: &str) -> f64 {
    let channel = |i: usize| {
        let c = u8::from_str_radix(&hex[i..i + 2], 16).unwrap() as f64 / 255.0;
        if c <= 0.03928 { c / 12.92 } else { ((c + 0.055) / 1.055).powf(2.4) }
    };
    0.2126 * channel(1) + 0.7152 * channel(3) + 0.0722 * channel(5)
}

#[derive(Debug)]
struct Swatch<'a> {
    name: &'a str,
    hex: &'a str,
}

impl Swatch<'_> {
    fn contrast(&self) -> f64 {
        let (a, b) = (luminance(self.hex), luminance(BG));
        (a.max(b) + 0.05) / (a.min(b) + 0.05)
    }
}

fn main() {
    let palette = [
        Swatch { name: "candle", hex: "#ECD8AE" },
        Swatch { name: "tide", hex: "#779BB0" },
        Swatch { name: "haze", hex: "#696D7D" },
    ];
    for swatch in &palette {
        // anything under 4.5:1 is recessive by design
        let note = if swatch.contrast() < 4.5 { "\trecessive" } else { "" };
        println!("{:<10}{:.1}:1{}", swatch.name, swatch.contrast(), note);
    }
}
contrast.go
package main

import (
	"fmt"
	"math"
	"strconv"
)

// BG is the editor background every ratio is measured against.
const BG = "#111214"

// luminance is relative luminance, as WCAG 2.1 defines it.
func luminance(hex string) float64 {
	channel := func(i int) float64 {
		v, _ := strconv.ParseUint(hex[i:i+2], 16, 8)
		c := float64(v) / 255
		if c <= 0.03928 {
			return c / 12.92
		}
		return math.Pow((c+0.055)/1.055, 2.4)
	}
	return 0.2126*channel(1) + 0.7152*channel(3) + 0.0722*channel(5)
}

type Swatch struct {
	Name string
	Hex  string
}

func (s Swatch) Contrast() float64 {
	a, b := luminance(s.Hex), luminance(BG)
	return (math.Max(a, b) + 0.05) / (math.Min(a, b) + 0.05)
}

func main() {
	palette := []Swatch{
		{"candle", "#ECD8AE"},
		{"tide", "#779BB0"},
		{"haze", "#696D7D"},
	}
	for _, s := range palette {
		// anything under 4.5:1 is recessive by design
		note := ""
		if s.Contrast() < 4.5 {
			note = "\trecessive"
		}
		fmt.Printf("%-10s%.1f:1%s\n", s.Name, s.Contrast(), note)
	}
}

const BG = "#111214";

/** Relative luminance, as WCAG 2.1 defines it. */
function luminance(hex: string): number {
  const channel = (i: number): number => {
    const c = parseInt(hex.slice(i, i + 2), 16) / 255;
    return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * channel(1) + 0.7152 * channel(3) + 0.0722 * channel(5);
}

class Swatch {
  constructor(readonly name: string, readonly hex: string) {}

  contrast(): number {
    const [a, b] = [luminance(this.hex), luminance(BG)];
    return (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
  }
}

import java.util.List;

public class Contrast {
    /** The editor background every ratio is measured against. */
    static final String BG = "#111214";

    /** Relative luminance, as WCAG 2.1 defines it. */
    static double luminance(String hex) {
        return 0.2126 * channel(hex, 1) + 0.7152 * channel(hex, 3) + 0.0722 * channel(hex, 5);
    }

    static double channel(String hex, int i) {
        double c = Integer.parseInt(hex.substring(i, i + 2), 16) / 255.0;
        return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
    }

    record Swatch(String name, String hex) {
        double contrast() {
            double a = luminance(this.hex), b = luminance(BG);
            return (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
        }
    }

    public static void main(String[] args) {
        var palette = List.of(
            new Swatch("candle", "#ECD8AE"),
            new Swatch("tide", "#779BB0"),
            new Swatch("haze", "#696D7D"));
        for (Swatch swatch : palette) {
            // anything under 4.5:1 is recessive by design
            String note = swatch.contrast() < 4.5 ? "\trecessive" : "";
            System.out.printf("%-10s%.1f:1%s%n", swatch.name(), swatch.contrast(), note);
        }
    }
}
